The canonical SaaS shell: top bar, a sidebar that folds from 236px to a 68px icon rail, and a fluid main area. The fold is w-59 → group-data-collapsed/shell:w-17, labels fade with group-data-collapsed/shell:opacity-0, and rail tooltips are generated by after:content-[attr(data-label)]. The script sets one attribute &mdash; it never touches a class &mdash; and now remembers the choice and answers the [ shortcut.

## JavaScript
```js
(() => {
  document.querySelectorAll('.tdl-02 [data-shell]').forEach((shell) => {
    if (shell.dataset.bound) return; shell.dataset.bound = '1';
    const btn = shell.querySelector('[data-toggle]');
    const set = (collapsed) => {
      shell.toggleAttribute('data-collapsed', collapsed);
      btn.setAttribute('aria-expanded', String(!collapsed));
      btn.setAttribute('aria-label', collapsed ? 'Expand navigation' : 'Collapse navigation');
      try { localStorage.setItem('tdl02-collapsed', collapsed ? '1' : '0'); } catch (e) {}
    };
    let saved = null; try { saved = localStorage.getItem('tdl02-collapsed'); } catch (e) {}
    if (saved === '1') set(true);
    btn.addEventListener('click', () => set(!shell.hasAttribute('data-collapsed')));
    document.addEventListener('keydown', (e) => {
      if (e.key !== '[' || e.metaKey || e.ctrlKey || e.altKey) return;
      if (e.target.closest && e.target.closest('input, textarea, select, [contenteditable]')) return;
      set(!shell.hasAttribute('data-collapsed'));
    });
  });
})();
```
