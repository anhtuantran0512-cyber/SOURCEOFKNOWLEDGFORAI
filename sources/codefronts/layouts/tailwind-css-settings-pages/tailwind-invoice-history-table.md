## JavaScript
```js
(() => {
  const root = document.querySelector('[data-tsp04]');
  if (!root || root.dataset.wired) return;
  root.dataset.wired = '1';
  const all = root.querySelector('[data-all]');
  const count = root.querySelector('[data-sel-count]');
  const rows = () => [...root.querySelectorAll('tbody tr')].filter((r) => r.offsetParent !== null);
  const sync = () => {
    const vis = rows();
    const on = vis.filter((r) => r.querySelector('[data-row-check]').checked).length;
    all.checked = vis.length > 0 && on === vis.length;
    all.indeterminate = on > 0 && on < vis.length;
    const total = root.querySelectorAll('[data-row-check]:checked').length;
    count.textContent = total + ' selected';
  };
  all.addEventListener('change', () => { rows().forEach((r) => { r.querySelector('[data-row-check]').checked = all.checked; }); sync(); });
  root.addEventListener('change', (e) => { if (e.target.matches('[data-row-check], [name=tsp-04-f]')) setTimeout(sync); });
  const save = (row) => {
    const lines = ['NORTHWIND CO — INVOICE', '', 'Invoice: ' + row.dataset.id, 'Status:  ' + row.dataset.status, 'Amount:  ' + row.dataset.amount, '', 'Thank you for your business.'];
    const url = URL.createObjectURL(new Blob([lines.join('\n')], { type: 'text/plain' }));
    const a = document.createElement('a');
    a.href = url; a.download = row.dataset.id + '.txt';
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  };
  const run = (btn) => {
    if (btn.dataset.state === 'loading') return;
    btn.dataset.state = 'loading';
    setTimeout(() => {
      save(btn.closest('tr'));
      btn.dataset.state = 'done';
      setTimeout(() => { btn.dataset.state = 'idle'; }, 1800);
    }, 900);
  };
  root.querySelectorAll('[data-dl]').forEach((b) => b.addEventListener('click', () => run(b)));
  root.querySelector('[data-bulk]').addEventListener('click', () => {
    root.querySelectorAll('[data-row-check]:checked').forEach((c, i) => setTimeout(() => run(c.closest('tr').querySelector('[data-dl]')), i * 250));
  });
  sync();
})();
```
